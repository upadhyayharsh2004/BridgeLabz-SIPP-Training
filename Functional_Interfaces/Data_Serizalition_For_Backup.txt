import java.io.*;

// Step 1: Marker interface
interface BackupSerializable {
    // no methods, just a marker
}

// Step 2: Classes that can be backed up
class Employee implements BackupSerializable, Serializable {
    private static final long serialVersionUID = 1L;
    private int id;
    private String name;

    public Employee(int id, String name) {
        this.id = id;
        this.name = name;
    }

    @Override
    public String toString() {
        return "Employee [ID=" + id + ", Name=" + name + "]";
    }
}

class Product implements BackupSerializable, Serializable {
    private static final long serialVersionUID = 1L;
    private String name;
    private double price;

    public Product(String name, double price) {
        this.name = name;
        this.price = price;
    }

    @Override
    public String toString() {
        return "Product [Name=" + name + ", Price=" + price + "]";
    }
}

// Step 3: Backup processor
class BackupProcessor {
    public static void backup(Object obj, String fileName) {
        if (!(obj instanceof BackupSerializable)) {
            System.out.println("Object is not marked for backup: " + obj.getClass().getSimpleName());
            return;
        }

        try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(fileName))) {
            oos.writeObject(obj);
            System.out.println("Backup successful for: " + obj.getClass().getSimpleName());
        } catch (IOException e) {
            System.err.println("Backup failed: " + e.getMessage());
        }
    }
}

// Step 4: Test the backup system
public class MarkerInterfaceDemo {
    public static void main(String[] args) {
        Employee emp = new Employee(101, "Alice");
        Product prod = new Product("Laptop", 1200.0);
        String message = "Hello"; // not marked for backup

        BackupProcessor.backup(emp, "employee_backup.dat");
        BackupProcessor.backup(prod, "product_backup.dat");
        BackupProcessor.backup(message, "message_backup.dat"); // should skip
    }
}
